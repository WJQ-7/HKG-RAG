\begin{algorithm}[H]
\caption{Hierarchical Knowledge Graph Construction}
\label{algo:hkg-construction}
\small
\begin{algorithmic}[1]

\Require Document collection $\mathcal{D}=\{d_i\}_{i=1}^{N}$
\Ensure Hierarchical knowledge graph $\mathcal{G}_{H}$

\State
$C_i
=
\operatorname{Chunk}(d_i)
=
\{c_{ij}\}_{j=1}^{M_i},
\quad 1\leq i\leq N$

\State
$E_D
=
\{e_D^i\}_{i=1}^{N}
=
\left\{
\operatorname{Merge}_{D}
\left(
\left\{
\operatorname{Extract}_{D}(c_{ij},name_{d_i})
\right\}_{j=1}^{M_i}
\right)
\right\}_{i=1}^{N}$

\State
$T_D
=
\{t_{d_i}\}_{i=1}^{N}
=
\left\{
\operatorname{Merge}_{T}
\left(
\left\{
\operatorname{Extract}_{T}(c_{ij},name_{d_i})
\right\}_{j=1}^{M_i}
\right)
\right\}_{i=1}^{N}$
\State
$(\hat{E}_C,\hat{R}_C)
=
\bigcup_{i=1}^{N}
\bigcup_{j=1}^{M_i}
\operatorname{Extract}_{C}(c_{ij})$

\State
$E_C
=
\operatorname{DeDup}(\hat{E}_C)$

\State
$R_C
=
\operatorname{Merge}_{R}
(\hat{R}_C,E_C)$

\State
$R_{DC}
=
\bigcup_{i=1}^{N}
\left\{
(e_D^i,r,e_C^k)
\mid
e_C^k\in E_C,\;
e_C^k\text{ occurs in }d_i
\right\}$

\State
$P_T
=
\{p_T^w\}_{w=1}^{W}
=
\operatorname{DBSCAN}
\left(
\left\{
\operatorname{Embed}(t_{d_i})
\right\}_{i=1}^{N}
\right),
\quad
\bigcup_{w=1}^{W}p_T^w
=
T_D$

\State
$E_T
=
\{e_T^w\}_{w=1}^{W}
=
\left\{
\operatorname{TopicSummary}(p_T^w)
\right\}_{w=1}^{W}$

\State
$R_{TD}
=
\bigcup_{w=1}^{W}
\left\{
(e_T^w,r,e_D^i)
\mid
t_{d_i}\in p_T^w
\right\}$

\State
$\mathcal{G}_{H}
=
\left(
E_C\cup E_D\cup E_T,\,
R_C\cup R_{DC}\cup R_{TD}
\right)$

\State \Return $\mathcal{G}_{H}$

\end{algorithmic}
\end{algorithm}
